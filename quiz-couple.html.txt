<script>
// QUESTIONS (18 – montée progressive d’audace)

  "Ressentez-vous encore du désir spontané ?",
  "Qui prend le plus souvent l’initiative du désir ?",
  "Avez-vous parfois envie de votre partenaire sans oser le montrer ?",
  "Avez-vous l’impression que le désir est réciproque ?",
  "Y a-t-il des envies que vous n’avez jamais osé exprimer ?",
  "Votre intimité est plutôt ?",
  "La tension entre vous est ?",
  "Aujourd’hui, votre couple vous donne surtout envie de ?",
  "Votre relation mérite davantage de…"
];

const choices = [
  ["Gérer le quotidien", "Partager un moment", "Se rapprocher"],
  ["Fonctionnelle", "Affectueuse", "Chargée de sous-entendus"],
  ["Rares", "Présents", "Spontanés"],
  ["Prévisible", "Calme", "Joueur"],
  ["Rapides", "Tendres", "Intenses"],
  ["Rarement", "Parfois", "Clairement"],
  ["Familier", "Réconfortant", "Désirable"],
  ["Difficile", "Possible", "Excitant"],
  ["Souvent", "Parfois", "Rarement"],
  ["Rarement", "Parfois", "Souvent"],
  ["Toujours la même personne", "Selon les moments", "Les deux"],
  ["Oui", "Parfois", "Non"],
  ["Pas vraiment", "À peu près", "Oui, clairement"],
  ["Beaucoup", "Quelques-unes", "Très peu"],
  ["Routinière", "Confortable", "Vivante"],
  ["Faible", "Variable", "Forte"],
  ["Sécurité", "Complicité", "Désir"],
  ["Stabilité", "Temps à deux", "Passion"]
];

let coupleCode = "";
let currentUser = null;
let currentQuestion = 0;
let answers = {};

// VALIDATION CODE
function validateCode() {
  coupleCode = document.getElementById("privateCode").value.trim();
  if (!coupleCode) return alert("Entre un code.");
  document.getElementById("stepCode").classList.add("hidden");
  document.getElementById("stepUser").classList.remove("hidden");
}

// DÉMARRER QUIZ
function startQuiz(user) {
  currentUser = user;
  currentQuestion = 0;
  answers = {};
  document.getElementById("stepUser").classList.add("hidden");
  document.getElementById("quiz").classList.remove("hidden");
  showQuestion();
}

// AFFICHER QUESTION
function showQuestion() {
  document.getElementById("questionBox").innerHTML = `
    <div class="progress">Question ${currentQuestion + 1}/${questions.length}</div>
    <h3>${questions[currentQuestion]}</h3>
    ${["A","B","C"].map((l,i)=>`
      <label>
        <input type="radio" name="answer" value="${l}">
        ${choices[currentQuestion][i]}
      </label>
    `).join("")}
  `;
}

// QUESTION SUIVANTE
function nextQuestion() {
  const selected = document.querySelector("input[name='answer']:checked");
  if (!selected) return alert("Choisis une réponse.");
  answers[currentQuestion] = selected.value;
  currentQuestion++;

  if (currentQuestion < questions.length) {
    showQuestion();
  } else {
    localStorage.setItem(`quiz_${coupleCode}_p${currentUser}`, JSON.stringify(answers));
    document.getElementById("quiz").classList.add("hidden");
    showResults();
  }
}

// PROFIL INDIVIDUEL
function profile(a) {
  let s = {A:0, B:0, C:0};
  Object.values(a).forEach(x => s[x]++);
  if (s.C > s.A && s.C > s.B) return "🔥 Plutôt amants";
  if (s.A > s.B && s.A > s.C) return "🧩 Plutôt partenaires";
  return "⚖️ Équilibre partenaires / amants";
}

// RÉSULTATS
function showResults() {
  const p1 = JSON.parse(localStorage.getItem(`quiz_${coupleCode}_p1`));
  const p2 = JSON.parse(localStorage.getItem(`quiz_${coupleCode}_p2`));

  let html = "";

  if (!p1 || !p2) {
    html = `
      <div class="card">
        <h4>⏳ En attente</h4>
        <p>L’autre partenaire doit maintenant répondre au questionnaire.</p>
        <p>Passez l’écran ou le lien à votre partenaire.</p>
      </div>
    `;
    document.getElementById("stepUser").classList.remove("hidden");
  } else {
    let gaps = Object.keys(p1).filter(i => p1[i] !== p2[i]);

    html = `
      <div class="card"><h4>👤 Partenaire 1</h4><p>${profile(p1)}</p></div>
      <div class="card"><h4>👤 Partenaire 2</h4><p>${profile(p2)}</p></div>

      <div class="card">
        <h4>🧠 Lecture de votre couple</h4>
        <p>
          ${
            gaps.length > 6
            ? "Vos réponses révèlent un décalage important autour du désir, du rythme ou de l’initiative."
            : "Vos réponses montrent une base commune solide, avec quelques nuances intéressantes dans l’expression du désir."
          }
        </p>
      </div>

      <div class="card">
        <h4>🔥 Conseil clé</h4>
        <p>
          Le désir ne disparaît pas par manque d’amour, mais par manque d’espace.
          Protégez des moments où vous n’êtes ni organisés, ni efficaces,
          mais simplement disponibles l’un pour l’autre.
        </p>
      </div>
    `;
  }

  const results = document.getElementById("results");
  results.innerHTML = html;
  results.classList.remove("hidden");
}

// RESET
function resetQuiz() {
  localStorage.clear();
  location.reload();
}
